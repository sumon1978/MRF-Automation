import os
from openpyxl import load_workbook
from copy import copy


# ==============================
# FILE PATH
# ==============================

solution_file = r"C:\Users\islam.munjurul\Desktop\MRF-Automation\5G\Solution file.xlsx"

template_file = r"C:\Users\islam.munjurul\Desktop\MRF-Automation\5G\Sample 5G MRF.xlsx"

output_folder = r"C:\Users\islam.munjurul\Desktop\MRF-Automation\5G"


# Create output folder
os.makedirs(output_folder, exist_ok=True)



# ==============================
# LOAD SOLUTION FILE
# ==============================

solution_wb = load_workbook(solution_file)

solution_ws = solution_wb.active



# ==============================
# LAST ROW
# ==============================

last_row = solution_ws.max_row



# ==============================
# COPY STYLE FUNCTION
# ==============================

def copy_sheet_style(source_ws, target_ws):

    for row in source_ws.iter_rows():

        for cell in row:

            new_cell = target_ws[cell.coordinate]

            if cell.has_style:
                new_cell._style = copy(cell._style)

            if cell.number_format:
                new_cell.number_format = cell.number_format

            if cell.alignment:
                new_cell.alignment = copy(cell.alignment)

            if cell.protection:
                new_cell.protection = copy(cell.protection)

            if cell.font:
                new_cell.font = copy(cell.font)

            if cell.fill:
                new_cell.fill = copy(cell.fill)

            if cell.border:
                new_cell.border = copy(cell.border)



# ==============================
# PROCESS EACH ROW
# ==============================

count = 0


for row in range(2, last_row + 1):


    # Read Solution Data

    site_id = solution_ws[f"A{row}"].value

    if site_id is None:
        continue


    address = solution_ws[f"B{row}"].value

    material = solution_ws[f"C{row}"].value

    contact = solution_ws[f"D{row}"].value



    # Load template

    template_wb = load_workbook(template_file)

    template_ws = template_wb.active



    # ==============================
    # INSERT DATA
    # ==============================


    # A -> G10

    template_ws["G10"] = site_id



    # B -> G9

    template_ws["G9"] = address



    # C -> G11

    template_ws["G11"] = material



    # D -> J9 and J11

    template_ws["J9"] = contact

    template_ws["J11"] = contact



    # ==============================
    # CREATE FILE NAME
    # ==============================


    site_name = str(site_id)

    c_name = str(material)



    # Clean filename

    site_name = (
        site_name
        .replace("/", "_")
        .replace("\\", "_")
        .replace(":", "_")
        .strip()
    )


    c_name = (
        c_name
        .replace("/", "_")
        .replace("\\", "_")
        .replace(":", "_")
        .strip()
    )



    # Final:
    # CGKTL39_MRF_UTSP.xlsx

    file_name = f"{site_name}_MRF_{c_name}.xlsx"



    output_file = os.path.join(
        output_folder,
        file_name
    )



    # Save

    template_wb.save(output_file)


    count += 1


    print("Created:", output_file)



# ==============================
# COMPLETE
# ==============================

print("==============================")
print("Completed Successfully")
print("Total Files Created:", count)
print("Output Location:")
print(output_folder)
print("==============================")
